< const canvas = document.createElement('canvas'); canvas.width = 400; canvas.height = 300; const ctx = canvas.getContext('2d'); const img = ctx.createImageData(400, 300); for (let y = 0; y < 300; y++) { for (let x = 0; x < 400; x++) { const nx = (x / 399) * 2 - 1; const ny = (y / 299) * 2 - 1; const r2 = nx * nx + ny * ny; const dx = nx * r2 * 0.25; const dy = ny * r2 * 0.25; const i = (y * 400 + x) * 4; img.data[i] = Math.max(0, Math.min(255, Math.round((0.5 + dx) * 255))); img.data[i + 1] = Math.max(0, Math.min(255, Math.round((0.5 + dy) * 255))); img.data[i + 2] = 0; img.data[i + 3] = 255; } } ctx.putImageData(img, 0, 0); document.getElementById('bulge').setAttribute('href', canvas.toDataURL());
